<script id="reviewsGQL" type="application/graphql">
query GetReviews($first: Int, $after: String) {
  reviews(first: $first, after: $after) {
    edges {
      node {
        rid
        text
        rating
        date
      }
      cursor
    }
    pageInfo {
      startCursor
      endCursor
      hasPreviousPage
      hasNextPage
    }
  }
}
</script>

<script>
async function loadPage(first) {
  const root = document.getElementById('latest-reviews');
  const spinner = document.getElementById('page-spinner');
  const loadMore = document.getElementById('page-load-more');
  spinner.classList.remove('d-none');
  loadMore.classList.add('d-none');

  const lastCursor = root.dataset.lastCursor;
  const query = document.getElementById('reviewsGQL').textContent;
  const firstPage = await fetch(
    "/api/graphql",
    {
      method: "POST",
      headers: {
        "Content-Type": "application/json",
        "Accept": "application/json",
      },
      body: JSON.stringify({
        query: query,
        variables: { first: first, after: lastCursor}
      })
    }
  )
  const template = document.getElementById('review-template').innerHTML;
  const data = (await firstPage.json()).data
  const edges = data.reviews.edges
  

  for (const edge of edges) {
    let element = document.createElement('div');
    element.innerHTML = nunjucks.renderString(template, {review: edge.node});
    root.appendChild(element.firstElementChild);
    root.dataset.lastCursor = edge.cursor;
  }
  spinner.classList.add('d-none');
  if (data.reviews.pageInfo.hasNextPage) {
    loadMore.classList.remove('d-none');
  }
}

document.addEventListener("DOMContentLoaded", async function() {
  await loadPage(20, null);
});

function openCodeOfConduct(url) {

}
