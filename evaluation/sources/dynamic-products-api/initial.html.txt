<script>
    document.addEventListener('DOMContentLoaded', function() {
        const productsUrl = '/ajax/products/json';

        function loadProducts() {
            fetch(productsUrl)
            .then(response => response.json())
            .then(data => {
                const container = document.getElementById('product-grid');
                container.innerHTML = '';

                data.forEach(product => {
                    const regex = /-(xs|s|m|l|xl|xxl|xxxl)-[a-z]+$/i;
                    let cleanName = product.name.replace(regex, '').trim().replace(/\s+/g, '-').toLowerCase();
                    const productUrl = `https://scrapingcourse.com/ecommerce/product/${cleanName}`;

                    const images = product.images.split(',');
                    const firstImage = images.length > 0 ? images[0] : 'path/to/default/image.png';
                    const imageName = firstImage.split('/').pop();
                    const newImageUrl = `https://scrapingcourse.com/ecommerce/wp-content/uploads/2024/03/${imageName}`;

                    const productHTML = `
                        <div
                            class="product-item flex flex-col items-center rounded-lg"
                            data-testid="product-item"
                            data-content="product-item"
                            itemscope
                            itemtype="http://schema.org/Product"
                        >
                            <a href="${productUrl}" class="product-link" data-testid="product-link" data-content="product-link">
                                <img
                                    class="product-image rounded-lg border shadow-md object-cover"
                                    width="200"
                                    height="240"
                                    decoding="async"
                                    fetchpriority="high"
                                    src="${newImageUrl}"
                                    alt="${product.name}" class="mb-2 rounded-md w-full object-cover"
                                    data-testid="product-image" data-content="product-image" itemprop="image"
                                >
                                <div class="product-info self-start text-left w-full" data-testid="product-info" data-content="product-info">
                                    <span  class="product-name" data-testid="product-name" data-content="product-name" itemprop="name">${product.name}</span>
                                    <br>
                                    <span
                                        class="product-price text-slate-600"
                                        data-testid="product-price"
                                        data-content="product-price"
                                        itemprop="priceCurrency"
                                        content="USD"
                                        itemprop="price"
                                    >$${product.regular_price}</span>
                                </div>
                            </a>
                            <meta itemprop="sku" content="MH01">
                            <meta itemprop="availability" content="http://schema.org/InStock">
                            <meta itemprop="itemCondition" content="http://schema.org/NewCondition">
                        </div>
                    `;
                    container.insertAdjacentHTML('beforeend', productHTML);
                });
            })
            .catch(error => {
                console.error("Error fetching data: ", error);
            });
        }

        loadProducts();
    });

</script>
